import httpx

async def transcribe_audio_stream(audio_bytes: bytes) -> str:
    api_url = "https://api.deepgram.com/v1/listen?model=nova-2&language=uz&smart_format=true"
    headers = {
        "Authorization": "Token DEEPGRAM_API_KEY_SHU_YERDA",
        "Content-Type": "audio/ogg"
    }
    async with httpx.AsyncClient() as client:
        response = await client.post(api_url, headers=headers, content=audio_bytes)
        if response.status_code == 200:
            return response.json()["results"]["channels"][0]["alternatives"][0]["transcript"]
        raise Exception("STT_API_GATEWAY_TIMEOUT")

import openai

async def execute_agent_brain(user_text: str, current_business_knowledge: str) -> str:
    system_prompt = f"""Siz Jarvis ko'p agentli tizimining bosh boshqaruvchisiz.
    Siz boshqarayotgan korxona qoidalari: {current_business_knowledge}
    Agar mijoz buyurtma bersa, javob oxiriga [ACTION:SAVE_ORDER] tegi qo'shilsin."""
    
    response = await openai.ChatCompletion.acreate(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_text}
        ],
        temperature=0.2
    )
    return response.choices[0].message.content
