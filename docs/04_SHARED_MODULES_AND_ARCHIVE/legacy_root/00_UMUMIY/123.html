async def ingress_webhook(request: Request, bg_tasks: BackgroundTasks):
    try:
        payload = await request.json()
        
        # 1. Tezkorlikni ta'minlash uchun og'ir ishni fonga olyapmiz

  • Shuning uchun audioni faqat operativ xotirada (RAM buffer, BytesIO) ushlash va Deepgram API ga yuborish shart.
import httpx
import io

async def process_voice_to_text(tg_file_bytes: bytes) -> str:
    # Deepgram API sozlamalari, O'zbek tili, eng oxirgi Nova-2 modeli
    api_url = "https://api.deepgram.com/v1/listen?model=nova-2&language=uz&smart_format=true"
    headers = {
        "Authorization": "Token KEYNI_SHU_YERGA_QO'YING",
        "Content-Type": "audio/ogg"
    }
    
    async with httpx.AsyncClient() as client:
        # Ovozli faylni xotiradan (RAM) stream ko'rinishida yuborish
        response = await client.post(api_url, headers=headers, content=tg_file_bytes)
        
        if response.status_code == 200:
            data = response.json()
            return data["results"]["channels"][0]["alternatives"][0]["transcript"]
        else:
            raise Exception(f"DEEPGRAM_STT_ERROR: Status {response.status_code}")

import openai

async def get_jarvis_decision(cleaned_text: str, business_rules: str) -> str:
    system_prompt = f"""Siz Tony Starkning Jarvis sun'iy intellekt tizimisiz.
    Siz hozir boshqarayotgan va avtomatlashtirayotgan korxona qoidalari va ma'lumotlari:
    {business_rules}
    
    Qoidalar:
    1. Foydalanuvchi bilan xuddi yaqin do'st va professional boshqaruvchi kabi o'zbek tilida gaplash.
    2. Agar mijoz buyurtma qilmoqchi bo'lsa yoki bazaga nimadir yozish kerak bo'lsa, javobingiz oxiriga [ACTION:SAVE_TO_DB] kalit so'zini qo'shing.
    """
    
    # Nol mablag' uchun eng tezkor va juda arzon model - gpt-4o-mini
    response = await openai.ChatCompletion.acreate(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": cleaned_text}
        ],
        temperature=0.3  # Fantziya qilmasdan aniq faktlar bilan ishlashi uchun
    )
    
    return response.choices[0].message.content
