# core/resilience.py

class CircuitBreaker:
    """
    3 holat:
      CLOSED    — normal, so'rovlar o'tadi
      OPEN      — xizmat yiqilgan, so'rov YUBORILMAYDI (tez xato)
      HALF_OPEN — sinov, 1 ta so'rov yuboriladi
    """
    def __init__(self, name, fail_threshold=5, timeout=60, success_threshold=2):
        self.name = name
        self.fail_threshold = fail_threshold      # 5 xato → OPEN
        self.timeout = timeout                    # 60s keyin HALF_OPEN
        self.success_threshold = success_threshold

    def call(self, fn, *args, fallback=None, **kwargs):
        state = self._get_state()                 # Redis'da

        if state == 'OPEN':
            if self._should_try(): self._set_state('HALF_OPEN')
            else:
                metrics.inc(f'cb.{self.name}.rejected')
                if fallback: return fallback()
                raise ServiceUnavailable(self.name)

        try:
            result = fn(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            metrics.inc(f'cb.{self.name}.failure')
            if fallback: return fallback()
            raise

"""

CHECKS = [
    ('orphan_jobs',    check_orphan_jobs,    heal_orphan_jobs),
    ('stuck_orders',   check_stuck_orders,   heal_stuck_orders),
    ('db_connections', check_db_conns,       heal_db_conns),
    ('redis_memory',   check_redis_mem,      heal_redis_mem),
    ('disk_space',     check_disk,           heal_disk),

    ('fiscal_queue',   check_fiscal_queue,   heal_fiscal_queue),
    ('ws_connections', check_ws,             heal_ws),
    ('cdn_health',     check_cdn,            heal_cdn),
    ('ai_providers',   check_ai,             heal_ai),
]

def run_cycle():
    for name, check, heal in CHECKS:
        try:
            problem = check()
            if not problem:
                metrics.gauge(f'health.{name}', 1)
                continue

            logger.warning(f"[HEAL] Muammo: {name} — {problem}")
            metrics.gauge(f'health.{name}', 0)
            metrics.inc(f'heal.{name}.attempt')

            healed = heal(problem)

            if healed:
                logger.info(f"[HEAL] ✅ Tuzatildi: {name}")
                metrics.inc(f'heal.{name}.success')
                notify_telegram(f"🔧 Avtomatik tuzatildi: {name}")
            else:
                logger.error(f"[HEAL] ❌ Tuzatib bo'lmadi: {name}")
                metrics.inc(f'heal.{name}.failed')
                alert_oncall(f"🚨 QO'LDA ARALASHUV KERAK: {name}\n{problem}")
        except Exception as e:
            logger.exception(f"[HEAL] Check xatosi: {name}")


# --- Namuna: orphan job ---
def check_orphan_jobs():
    """Worker o'lganda `running` holatda qolgan job'lar"""
    cutoff = timezone.now() - timedelta(minutes=15)
    orphans = Job.objects.filter(status='running', heartbeat_at__lt=cutoff)
    return list(orphans) if orphans.exists() else None

def heal_orphan_jobs(orphans):
    for job in orphans:
        if job.attempts < job.max_attempts:
            job.status, job.attempts = 'queued', job.attempts + 1
            job.save(update_fields=['status', 'attempts'])
            enqueue(job.queue, job.id)
        else:
            job.status = 'failed'; job.save()
            move_to_dlq(job)
    return True

ALLOWED = {
    'new':      {'accepted', 'cancelled'},
    'accepted': {'cooking', 'cancelled'},
    'cooking':  {'ready', 'cancelled'},
    'ready':    {'served', 'cancelled'},
    'served':   {'paid', 'cancelled'},
    'paid':     set(),
    'cancelled':set(),
}

def transition(order, new_status, actor=None):
    if new_status not in ALLOWED[order.status]:
        raise InvalidTransition(f"{order.status} → {new_status} mumkin emas")
    with transaction.atomic():
        old = order.status
        order.status = new_status
        setattr(order, f'{new_status}_at', timezone.now())
        order.save(update_fields=['status', f'{new_status}_at', 'updated_at'])
        AuditLog.log(order, old, new_status, actor)
    transaction.on_commit(lambda: broadcast_order_status(order))

def create_order(session, table_id, items, note, idem_key):
    # 1. Idempotency (Redis, TTL 24s)
    existing = check_idempotency(idem_key)
    if existing: return existing                          # dublikat yo'q ✅

    # 2. Stol tekshiruvi (tenant scope!)
    table = Table.objects.select_related('restaurant').get(
        id=table_id, restaurant_id=session.restaurant_id, is_active=True)

    # 3. Restoran faolmi
    if not table.restaurant.is_active: raise RestaurantInactive()

    # 4. Taomlar tekshiruvi (BITTA so'rov, N+1 yo'q)
    dish_ids = [i['dish_id'] for i in items]
    dishes = {d.id: d for d in Dish.objects.filter(
        id__in=dish_ids, restaurant_id=session.restaurant_id
    ).prefetch_related('modifiers')}

    if len(dishes) != len(set(dish_ids)): raise TenantMismatch()

    # 5. Mavjudlik (stop-list)
    for i in items:
        d = dishes[i['dish_id']]
        if not d.is_available: raise DishUnavailable(d)

    # 6. ⚡ NARX SERVERDA HISOBLANADI (frontendga ISHONILMAYDI)
    subtotal = Decimal(0)
    order_items = []
    for i in items:
        d = dishes[i['dish_id']]
        qty = max(1, min(99, int(math.ceil(i['qty']))))    # butun son ⚡
        mod_sum = sum(m.price_delta for m in d.modifiers.filter(id__in=i['modifiers']))
        line = (d.price + mod_sum) * qty
        subtotal += line
        order_items.append(OrderItem(dish=d, qty=qty, price=d.price + mod_sum,
                                     modifiers=i['modifiers'], note=i['note'][:200],
                                     ar_viewed=i.get('ar_viewed', False)))

    service_fee = subtotal * table.restaurant.service_fee_pct / 100
    total = subtotal + service_fee

    # 7. Transaksiya
    with transaction.atomic():
        order = Order.objects.create(
            restaurant_id=session.restaurant_id, table=table, guest=session.guest,
            session_id=session.id, number=next_number(table.restaurant),
            subtotal=subtotal, service_fee=service_fee, total=total,
            note=note[:500], eta_min=max(d.cook_min for d in dishes.values()),
            idempotency_key=idem_key, source='qr',
        )
        for oi in order_items: oi.order = order
        OrderItem.objects.bulk_create(order_items)
        table.status = 'ordering'; table.save(update_fields=['status'])

    # 8. Post-commit (transaksiya muvaffaqiyatli bo'lgandan KEYIN)
    transaction.on_commit(lambda: _after_order_created(order))

    save_idempotency(idem_key, order)
    return order

class ReliableWS {
  constructor(url) {
    this.url = url; this.retries = 0; this.lastSeq = 0;
    this.connect();
  }
  connect() {
    this.ws = new WebSocket(this.url);
    this.ws.onopen = () => {
      this.retries = 0;
      this.startHeartbeat();
      this.syncMissed();                    // ⚡ o'tkazib yuborilganlarni olish
    };
    this.ws.onmessage = (e) => {
      const m = JSON.parse(e.data);
      if (m.seq && m.seq <= this.lastSeq) return;   // dublikat
      this.lastSeq = m.seq || this.lastSeq;
      this.handle(m);
    };
    this.ws.onclose = () => {
      this.stopHeartbeat();
      const delay = Math.min(1000 * 2 ** this.retries, 30000);  // exponential
      const jitter = delay * (0.8 + Math.random() * 0.4);
      this.retries++;
      setTimeout(() => this.connect(), jitter);
      if (this.retries > 3) this.startPolling();    // ⚡ fallback
    };
  }
  async syncMissed() {
    // WS uzilgan paytdagi o'zgarishlarni REST orqali olish
    const r = await fetch(`/v1/sync/?since_seq=${this.lastSeq}`);
    (await r.json()).events.forEach(m => this.handle(m));
  }
  startPolling() { /* har 10s REST so'rov */ }
}

def t(obj, field: str, lang: str = 'uz') -> str:
    """
    t(dish, 'name', 'ru') → dish.name_ru yoki fallback dish.name_uz
    """
    val = getattr(obj, f'{field}_{lang}', None)
    if val: return val
    for fb in ('uz', 'ru', 'en'):                 # fallback zanjiri
        val = getattr(obj, f'{field}_{fb}', None)
        if val: return val
    return ''

import subprocess, json, os

def optimize_glb(input_path: str, output_path: str, real_size_cm: float) -> dict:
    """
    1. Draco siqish  2. Tekstura 1024px  3. Poligon ≤60k
    4. ⚡ Real o'lchamga masshtablash  5. Markazni Y=0 ga
    """
    tmp = output_path + '.tmp.glb'

    # gltf-transform pipeline
    cmds = [
        ['gltf-transform', 'resize', input_path, tmp, '--width','1024','--height','1024'],
        ['gltf-transform', 'simplify', tmp, tmp, '--ratio','0.5','--error','0.001'],
        ['gltf-transform', 'webp', tmp, tmp, '--quality','85'],
        ['gltf-transform', 'draco', tmp, output_path],
    ]
    for c in cmds:
        subprocess.run(c, check=True, timeout=300)

    # ⚡ Real o'lchamga masshtablash va markazlash (pygltflib)
    scale_and_center(output_path, real_size_cm / 100.0)   # sm → m

    if size > 3 * 1024 * 1024:                            # 3 MB dan katta
        aggressive_optimize(output_path)                   # qayta, agressiv
        size = os.path.getsize(output_path)

    return {'bytes': size, 'ok': size <= 3*1024*1024}

def scale_and_center(glb_path: str, target_size_m: float):
    """
    Model AR'da 1:1 chiqishi uchun:
      1. Bounding box hisoblanadi
      2. Eng katta o'lcham target_size_m ga tenglashtiriladi
      3. Model tagi Y=0 ga tushiriladi (stolga to'g'ri qo'yiladi)
    """
    gltf = GLTF2().load(glb_path)
    bbox = compute_bbox(gltf)
    current_max = max(bbox.size.x, bbox.size.z)   # gorizontal o'lcham
    scale = target_size_m / current_max

    root = gltf.nodes[gltf.scenes[0].nodes[0]]
    root.scale = [scale, scale, scale]
    root.translation = [
        -bbox.center.x * scale,
        -bbox.min.y   * scale,      # ⚡ tagi Y=0
        -bbox.center.z * scale,
    ]
    gltf.save(glb_path)

// static/guest/js/i18n.js
let TRANSLATIONS = {};
let CURRENT_LANG = 'uz';

async function loadI18n(lang) {
  CURRENT_LANG = lang;
  const cached = localStorage.getItem(`i18n_${lang}_v${I18N_VERSION}`);
  if (cached) { TRANSLATIONS = JSON.parse(cached); return; }

  const r = await fetch(`/static/i18n/${lang}.json?v=${I18N_VERSION}`);
  TRANSLATIONS = await r.json();
  localStorage.setItem(`i18n_${lang}_v${I18N_VERSION}`, JSON.stringify(TRANSLATIONS));
}

/**
 * t('cart.empty')            → "Savat bo'sh"
 * t('order.eta', {n: 18})    → "Taxminan 18 daqiqa qoldi"
 */
function t(key, params = {}) {
  const val = key.split('.').reduce((o, k) => o?.[k], TRANSLATIONS);
  if (val === undefined) { console.warn(`[i18n] Yo'q: ${key}`); return key; }
  return val.replace(/\{(\w+)\}/g, (_, p) => params[p] ?? `{${p}}`);
}

/** Sahifadagi barcha data-i18n elementlarini tarjima qilish */
function applyI18n(root = document) {
  root.querySelectorAll('[data-i18n]').forEach(el => {
    el.textContent = t(el.dataset.i18n);
  });
  root.querySelectorAll('[data-i18n-ph]').forEach(el => {
    el.placeholder = t(el.dataset.i18nPh);
  });
  root.querySelectorAll('[data-i18n-title]').forEach(el => {
    el.title = t(el.dataset.i18nTitle);
  });
  document.documentElement.lang = CURRENT_LANG;
}

/** Til almashtirish (sahifa qayta yuklanmasdan) */

}

TRANSLATE_PROMPT = """Sen professional tarjimonsan. O'zbek restorani menyusini tarjima qilyapsan.

QOIDALAR:
1. Milliy o'zbek taomlari nomini TARJIMA QILMA — transliteratsiya qil.
   Masalan: "Norin" → en: "Norin", ru: "Норин"
2. Tavsifda taom nima ekanini QISQA tushuntir (chet ellik bilmaydi).
3. Umumiy taomlar tarjima qilinadi: "Salat" → "Salad" / "Салат"
4. Ohang: iliq, ishtaha ochadigan, lekin ortiqcha emas.
5. Faqat JSON qaytar, boshqa hech narsa.

Tarjima qilinadigan (o'zbekchadan {target_lang} ga):
{items}

Format:
[{"id": 15, "name": "...", "desc": "..."}]
"""

def auto_translate_dishes(restaurant_id: int, target_lang: str, batch_size: int = 40):
    field_name = f'name_{target_lang}'
    field_desc = f'desc_{target_lang}'

    qs = Dish.objects.filter(restaurant_id=restaurant_id).filter(
        Q(**{f'{field_name}': ''}) | Q(**{f'{field_desc}': ''})
    )[:batch_size]

    if not qs: return {'translated': 0}

    items = [{'id': d.id, 'name': d.name_uz, 'desc': d.desc_uz} for d in qs]

    # ⚡ MCP orqali (Qism VII) — fallback zanjiri bilan
    result = mcp_router.complete(
        task='translate',
        prompt=TRANSLATE_PROMPT.format(target_lang=LANG_NAMES[target_lang],
                                       items=json.dumps(items, ensure_ascii=False)),
        json_mode=True,
        restaurant_id=restaurant_id,
    )

    updated = 0
    for item in result:
        d = next((x for x in qs if x.id == item['id']), None)
        if not d: continue
        if not getattr(d, field_name): setattr(d, field_name, item['name'])
        if not getattr(d, field_desc): setattr(d, field_desc, item['desc'])
        d.tr_auto[target_lang] = True             # ⚡ AI tarjimasi belgisi
        d.tr_approved[target_lang] = False        # admin tasdiqlashi kerak
        d.save()
        updated += 1

    bump_menu_version(restaurant_id)
    return {'translated': updated}

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any

@dataclass
class MCPRequest:
    task: str                      # chat|translate|vision|image_to_3d|embedding
    prompt: str = ''
    system: str = ''
    messages: list = None
    images: list = None            # rasm URL yoki base64
    json_mode: bool = False
    max_tokens: int = 1500
    temperature: float = 0.3
    timeout: int = 30
    extra: dict = None

@dataclass
class MCPResponse:
    ok: bool
    content: Any = None
    provider: str = ''
    model: str = ''
    tokens_in: int = 0
    tokens_out: int = 0
    cost_usd: float = 0.0
    latency_ms: int = 0
    error: str = ''
    raw: dict = None


class MCPProvider(ABC):
    """Har bir AI provayder shu interfeysni amalga oshiradi."""
    name: str
    supported_tasks: set
    priority: dict          # {'chat': 10, 'translate': 20}
    price_in: float         # $/1M token
    price_out: float

    @abstractmethod
    def is_configured(self) -> bool:
        """API key mavjudmi"""

    @abstractmethod
    def call(self, req: MCPRequest) -> MCPResponse:
        """Asosiy chaqiruv"""

    def health_check(self) -> bool:
        """Provayder tirikmi (yengil so'rov)"""
        try:
            r = self.call(MCPRequest(task='chat', prompt='ping', max_tokens=5, timeout=8))
            return r.ok
        except Exception:
            return False

    def estimate_cost(self, tin: int, tout: int) -> float:
        return (tin * self.price_in + tout * self.price_out) / 1_000_000

    priority = {'chat': 20, 'translate': 10, 'vision': 15, 'reasoning': 5}
    price_in, price_out = 0.80, 4.00
    MODEL = 'claude-haiku-4-5-20251001'

    def __init__(self):
        self.key = settings.AI_KEYS.get('ANTHROPIC_API_KEY', '')
        self.client = anthropic.Anthropic(api_key=self.key) if self.key else None

    def is_configured(self): return bool(self.key)

    def call(self, req: MCPRequest) -> MCPResponse:
        t0 = time.time()
        try:
            msgs = req.messages or [{'role':'user','content': req.prompt}]
            if req.images:
                msgs[-1]['content'] = [
                    {'type':'image','source':{'type':'url','url': u}} for u in req.images
                ] + [{'type':'text','text': req.prompt}]

            kwargs = dict(model=self.MODEL, max_tokens=req.max_tokens,
                          temperature=req.temperature, messages=msgs)
            if req.system: kwargs['system'] = req.system

            resp = self.client.messages.create(**kwargs, timeout=req.timeout)
            text = resp.content[0].text
            content = json.loads(_strip_fences(text)) if req.json_mode else text

            return MCPResponse(
                ok=True, content=content, provider=self.name, model=self.MODEL,
                tokens_in=resp.usage.input_tokens, tokens_out=resp.usage.output_tokens,
                cost_usd=self.estimate_cost(resp.usage.input_tokens, resp.usage.output_tokens),
                latency_ms=int((time.time()-t0)*1000),
            )
        except Exception as e:
            return MCPResponse(ok=False, provider=self.name, error=str(e),
                               latency_ms=int((time.time()-t0)*1000))

            # 1. Vazifa yaratish

                'image_url': req.images[0],
                'enable_pbr': True,
                'should_remesh': True,
                'target_polycount': 40000,
                'topology': 'triangle',

            task_id = r.json()['result']

            while time.time() < deadline:
                time.sleep(10)
                s = requests.get(f'{self.BASE}/image-to-3d/{task_id}', headers=h, timeout=20).json()
                st = s.get('status')
                if st == 'SUCCEEDED':
                    return MCPResponse(
                        ok=True, provider=self.name, model='meshy-4',
                        content={'glb_url': s['model_urls']['glb'],
                                 'thumbnail': s.get('thumbnail_url')},
                        cost_usd=self.COST_PER_MODEL,
                        latency_ms=int((time.time()-t0)*1000))
                if st in ('FAILED', 'EXPIRED'):
                    return MCPResponse(ok=False, provider=self.name,
                                       error=s.get('task_error', {}).get('message', st))
            return MCPResponse(ok=False, provider=self.name, error='timeout')

            return MCPResponse(ok=False, provider=self.name, error=str(e))

# apps/ai/mcp/router.py
import hashlib, json, logging
from django.core.cache import cache
from django.conf import settings
from core.resilience import CircuitBreaker
from .base import MCPRequest, MCPResponse
from .registry import ALL_PROVIDERS

log = logging.getLogger('ai.mcp')

class MCPRouter:
    """
    Vazifani mos provayderga yo'naltiradi.
    Yiqilsa — keyingisiga. Xarajatni hisoblaydi. Keshlaydi.
    """
    def __init__(self):
        self.providers = [p() for p in ALL_PROVIDERS]
        self.breakers = {p.name: CircuitBreaker(f'ai.{p.name}',
                          fail_threshold=3, timeout=120) for p in self.providers}

    # ---- Asosiy metod ----
    def run(self, req: MCPRequest, restaurant_id=None, use_cache=True) -> MCPResponse:
        # 1. Kesh
        ck = self._cache_key(req)
        if use_cache and req.task in ('translate', 'chat'):
            cached = cache.get(ck)
            if cached:
                log.debug(f'[MCP] kesh: {req.task}')
                return MCPResponse(**cached)

        # 2. Byudjet
        if restaurant_id and not self._check_budget(restaurant_id):
            return MCPResponse(ok=False, error='AI_BUDGET_EXCEEDED')

        # 3. Provayder ro'yxati (prioritet bo'yicha)
        candidates = self._candidates(req.task)
        if not candidates:
            return MCPResponse(ok=False, error=f'NO_PROVIDER_FOR_{req.task}')

        # 4. Zanjir bo'ylab urinish
        errors = []
        for p in candidates:
            br = self.breakers[p.name]
            if br.is_open():
                errors.append(f'{p.name}: circuit_open')
                continue

            log.info(f'[MCP] {req.task} → {p.name}')
            resp = p.call(req)

            if resp.ok:
                br.on_success()
                self._record(resp, restaurant_id, req.task)
                if use_cache and req.task in ('translate', 'chat'):
                    cache.set(ck, resp.__dict__, 3600)
                return resp

            br.on_failure()
            errors.append(f'{p.name}: {resp.error}')
            log.warning(f'[MCP] {p.name} xato: {resp.error} → keyingisi')

        log.error(f'[MCP] BARCHA provayder yiqildi: {errors}')
        return MCPResponse(ok=False, error='ALL_PROVIDERS_FAILED',
                           raw={'errors': errors})

    # ---- Yordamchilar ----
    def _candidates(self, task):
        ps = [p for p in self.providers
              if p.is_configured() and task in p.supported_tasks]
        return sorted(ps, key=lambda p: p.priority.get(task, 999))

    def _cache_key(self, req):
        raw = f'{req.task}|{req.system}|{req.prompt}|{req.temperature}'
        return f'mcp:{hashlib.sha256(raw.encode()).hexdigest()[:32]}'

    def _check_budget(self, restaurant_id):
        today = timezone.now().date()
        spent = AIUsage.objects.filter(
            restaurant_id=restaurant_id, created_at__date=today
        ).aggregate(s=Sum('cost_usd'))['s'] or 0
        limit = settings.AI_DAILY_LIMIT_PER_RESTAURANT   # masalan $2
        if spent >= limit:
            log.warning(f'[MCP] Byudjet tugadi: rest={restaurant_id} ${spent}')
            return False
        return True

    def _record(self, resp, restaurant_id, task):
        AIUsage.objects.create(
            restaurant_id=restaurant_id, provider=resp.provider, model=resp.model,
            task=task, tokens_in=resp.tokens_in, tokens_out=resp.tokens_out,
            cost_usd=resp.cost_usd, latency_ms=resp.latency_ms)
        metrics.inc(f'ai.{resp.provider}.success')
        metrics.observe('ai.latency_ms', resp.latency_ms, {'provider': resp.provider})
        metrics.add('ai.cost_usd', resp.cost_usd, {'provider': resp.provider})


mcp = MCPRouter()      # singleton

def build_menu_context(restaurant_id: int, lang: str) -> str:
    """
    To'liq menyu emas — faqat AI uchun zarur maydonlar.
    100 taom ≈ 1500 token (to'liq bo'lsa 8000+ bo'lardi).
    """
    key = f'ai:menuctx:{restaurant_id}:{lang}'
    ctx = cache.get(key)
    if ctx: return ctx

    dishes = Dish.objects.filter(
        restaurant_id=restaurant_id, is_available=True
    ).select_related('category').only(
        'id','name_uz','name_ru','name_en','price','diet_tags','allergens',
        'cook_min','category__name_uz','category__name_ru','category__name_en')

    lines = []
    for d in dishes:
        tags = ','.join(d.diet_tags) if d.diet_tags else '-'
        alrg = ','.join(d.allergens) if d.allergens else '-'
        lines.append(f"{d.id}|{t(d,'name',lang)}|{int(d.price)}|{tags}|{alrg}|{d.cook_min}min")

    ctx = "ID|Nom|Narx|Teglar|Allergen|Vaqt\n" + "\n".join(lines)
    cache.set(key, ctx, 300)
    return ctx
